 28 // Function definition for sorting
 29 void Sort(int arr[], int n) {
 30     int temp;
 31     for (int i = 0; i < n - 1; i++) {
 32         for (int j = 0; j < n - i - 1; j++) {
 33             // Swap if the element found is greater than the next element
 34             if (arr[j] > arr[j + 1]) {
 35                 temp = arr[j];
 36                 arr[j] = arr[j + 1];
 37                 arr[j + 1] = temp;
 38             }
 39         }
 40     }
 41 }
